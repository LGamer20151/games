import { GoogleGenAI } from "@google/genai"; // Initialize the API with your secret key const ai = new GoogleGenAI({ apiKey: process.env.GEMINI_API_KEY }); async function askAIWithGoogle(userQuery) { const response = await ai.models.generateContent({ model: 'gemini-2.0-flash', // A fast, powerful model contents: userQuery, config: { // This line tells Gemini to use Google Search before answering! tools: [{ googleSearch: {} }], }, }); // This is the final text answer backed by Google Search return response.text; }
