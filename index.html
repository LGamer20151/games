import java.util.*; public class ChatBot { // Simple pair of pattern and response templates private static class Rule { final String pattern; final String[] responses; Rule(String pattern, String... responses) { this.pattern = pattern.toLowerCase(); this.responses = responses; } } private final List rules = new ArrayList<>(); private final Random random = new Random(); public ChatBot() { // Greeting addRule("hi|hello|hey", "Hello! How can I help you today?", "Hi there! What would you like to talk about?"); // Ask about needs addRule("how are you|how's it going", "I'm just a bunch of code, but I'm here to help you. What would you like to do?", "I'm running smoothly. How can I assist you?"); // About Java addRule("java.*(version|versions)?", "I can help you with Java basics, or compare versions. What would you like to know?", "Java is a versatile language running on the JVM. Need a quick guide?"); // Coding help addRule("how to|give me a code|code for|write (a|an) (program|function)", "Sure, tell me the goal and the language. If you want, I can provide a starter example.", "Give me a task description and I will draft a sample implementation."); // Welcome to AI demo addRule("ai|artificial intelligence|machine learning", "AI can mean many things. Do you want a simple chatbot, a game AI, or a data model example?", "Are you looking for a basic AI for a game or a web service?"); // Default fallback addRule(".*", "Tell me more about what you want to build with Java. I can provide code samples for a chatbot, a game AI, or a REST service."); } private void addRule(String pattern, String... responses) { rules.add(new Rule(pattern, responses)); } private String respond(String input) { String in = input.trim().toLowerCase(); for (Rule r : rules) { if (in.matches(".*" + r.pattern + ".*")) { return r.responses[random.nextInt(r.responses.length)]; } } // Fallback return "I'm not sure I understood that. Could you rephrase or share more details?"; } public static void main(String[] args) { ChatBot bot = new ChatBot(); Scanner scanner = new Scanner(System.in); System.out.println("ChatBot: Hi! I can help with Java code examples. Type 'exit' to quit."); while (true) { System.out.print("You: "); String userInput = scanner.nextLine(); if (userInput == null) continue; if (userInput.equalsIgnoreCase("exit") || userInput.equalsIgnoreCase("quit")) { System.out.println("ChatBot: Goodbye!"); break; } String reply = bot.respond(userInput); System.out.println("ChatBot: " + reply); } scanner.close(); } }
